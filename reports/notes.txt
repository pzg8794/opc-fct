	@SuppressWarnings("unused")
	private int getOpenS(char[] exp, char symbol) {
		int i = 0;
		int answer = 0;

		for(Character ch: exp){

			i++;
			if( ch == symbol)

		}

		if( answer == 0)
			return -1;

		return answer;
	}
